void PollServer::run() {
  std::cout << "> calculator_server: started, listen fd = " << _listen_fd << std::endl;
  while (true) {
    int ready = poll(_poll_fds.data(), _poll_fds.size(), -1);
    if (ready < 0) {
      if (errno == EINTR) {
        continue;
      }
      throw std::runtime_error(errno_message("poll(2) error"));
    }

    if (!_poll_fds.empty() && (_poll_fds[0].revents & POLLIN)) {
      accept_clients();
    }

    for (std::size_t i = _clients.size(); i > 0u; --i) {
      std::size_t index = i - 1u;
      short events = _poll_fds[index + 1u].revents;
      if (events & (POLLERR | POLLHUP | POLLNVAL)) {
        close_client(index);
        continue;
      }
      if (events & POLLIN) {
        read_client(index);
      }
      if (index < _clients.size() && (events & POLLOUT)) {
        write_client(index);
      }
    }

    rebuild_events();
  }
}

std::string CalculatorModel::execute(const std::string& user, const std::string& command) {
  if (!valid_user_name(user)) {
    throw CalculatorError("некорректное имя пользователя");
  }

  Variables backup = _vars[user];
  Variables& current = _vars[user];
  std::string text = protocol::trim(command);
  if (text.empty()) {
    throw CalculatorError("пустая команда");
  }

  if (text == "показать" || text == "show") {
    return variables_to_string(current);
  }

  const std::string calc_ru = "вычислить";
  const std::string calc_en = "calc";
  if (protocol::starts_with(text, calc_ru)) {
    std::string expression = protocol::trim(text.substr(calc_ru.size()));
    long long value = evaluate_expression(expression, current);
    return std::to_string(value);
  }
  if (protocol::starts_with(text, calc_en)) {
    std::string expression = protocol::trim(text.substr(calc_en.size()));
    long long value = evaluate_expression(expression, current);
    return std::to_string(value);
  }

  std::size_t eq = text.find('=');
  if (eq != std::string::npos) {
    char variable = parse_assigned_variable(text.substr(0u, eq));
    std::string expression = text.substr(eq + 1u);
    long long value = evaluate_expression(expression, current);
    if (variable == 'X') {
      current.x = value;
    } else if (variable == 'Y') {
      current.y = value;
    } else {
      current.z = value;
    }

    try {
      save();
    } catch (...) {
      current = backup;
      throw;
    }

    std::string var_name(1u, variable);
    return var_name + "=" + std::to_string(value);
  }

  throw CalculatorError("неизвестная команда: используйте 'X = выражение', 'вычислить выражение' или 'показать'");
}

void PollServer::process_lines(std::size_t index) {
  Client& client = _clients[index];

void PollServer::run() {
  std::cout << "> calculator_server: started, listen fd = " << _listen_fd << std::endl;
  while (true) {
    int ready = poll(_poll_fds.data(), _poll_fds.size(), -1);
    if (ready < 0) {
      if (errno == EINTR) {
        continue;
      }
      throw std::runtime_error(errno_message("poll(2) error"));
    }

    if (!_poll_fds.empty() && (_poll_fds[0].revents & POLLIN)) {
      accept_clients();
    }

    for (std::size_t i = _clients.size(); i > 0u; --i) {
      std::size_t index = i - 1u;
      short events = _poll_fds[index + 1u].revents;
      if (events & (POLLERR | POLLHUP | POLLNVAL)) {
        close_client(index);
        continue;
      }

  addrinfo hints;
  std::memset(&hints, 0, sizeof(hints));
  hints.ai_family = AF_UNSPEC;
  hints.ai_socktype = SOCK_STREAM;

  addrinfo* result = nullptr;
  int code = getaddrinfo(host.c_str(), port.c_str(), &hints, &result);
  if (code != 0) {
    throw std::runtime_error(std::string("getaddrinfo: ") + gai_strerror(code));
  }

  for (addrinfo* item = result; item != nullptr; item = item->ai_next) {
    _socket = socket(item->ai_family, item->ai_socktype, item->ai_protocol);
    if (_socket < 0) {
      continue;
    }
    if (connect(_socket, item->ai_addr, item->ai_addrlen) == 0) {
      break;
    }
    close(_socket);
    _socket = -1;
  }
  freeaddrinfo(result);

void PollServer::rebuild_events() {
  _poll_fds.clear();

  pollfd listener;
  listener.fd = _listen_fd;
  listener.events = POLLIN;
  listener.revents = 0;
  _poll_fds.push_back(listener);

      skip_spaces();
      if (match('+')) {
        value += parse_term();
      } else if (match('-')) {
        value -= parse_term();
      } else {
        break;
      }
    }

  }

    long long value = parse_factor();
    while (true) {
      skip_spaces();

        value *= parse_factor();

        }

      } else {
        break;
      }

      return parse_factor();
    }
    if (match('-')) {
      return -parse_factor();

      long long value = parse_expression();
      skip_spaces();
      if (!match(')')) {
        throw CalculatorError("не закрыта скобка");

      return value;

      ++_pos;
      return _variables.y;
    }

      return _variables.z;

Как различить программу, процесс и поток на примере работы?

Программа - исполняемый код, процесс - запущенный экземпляр с ресурсами, поток - последовательность исполнения внутри него. Один файл клиента можно запустить несколько раз и получить несколько процессов. У сервера в этой реализации один прикладной поток и много сокетов.

Почему ошибочные ответы неверны: Это три названия одного TCP-сокета. Сокет является отдельным ресурсом процесса. Каждый метод класса автоматически является потоком. Методы исполняются тем потоком, который их вызывает.
